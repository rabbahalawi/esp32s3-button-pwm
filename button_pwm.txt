#include <Arduino.h>

// ---------- Constants (ESP32-S3 pins) ----------
const uint8_t BUTTON_PIN     = 5;
const uint8_t POT_PIN        = 4;
const uint8_t STATUS_LED_PIN = 6;
const uint8_t PWM_LED_PIN    = 7;

// ---------- State ----------
bool buttonPressed = false;
bool pwmReady      = false;
int  rawInput      = 0;
int  requestedDuty = 0;
int  appliedDuty   = 0;

// ---------- Prototypes ----------
void readInputs();
void processInputs();
void updateOutputs();
int  scaleToDuty(int raw);

void setup() {
  Serial.begin(115200);
  pinMode(BUTTON_PIN, INPUT_PULLUP);
  pinMode(STATUS_LED_PIN, OUTPUT);
  digitalWrite(STATUS_LED_PIN, LOW);
  analogReadResolution(12);
  analogSetPinAttenuation(POT_PIN, ADC_11db);

  pwmReady = ledcAttach(PWM_LED_PIN, 5000, 8);
  if (pwmReady) {
    ledcWrite(PWM_LED_PIN, 0);
  } else {
    Serial.println("PWM setup failed.");
  }
}

void loop() {
  readInputs();
  processInputs();
  updateOutputs();

  Serial.printf("btn=%d raw=%d req=%d applied=%d\n",
                buttonPressed, rawInput, requestedDuty, appliedDuty);

  delay(20);
}

// ---------- INPUT ----------
void readInputs() {
  buttonPressed = (digitalRead(BUTTON_PIN) == LOW);
  rawInput = analogRead(POT_PIN);
}

// ---------- PROCESSING ----------
int scaleToDuty(int raw) {
  return constrain(map(raw, 0, 4095, 0, 255), 0, 255);
}

void processInputs() {
  requestedDuty = scaleToDuty(rawInput);
  appliedDuty = (buttonPressed && pwmReady) ? requestedDuty : 0;
}

// ---------- OUTPUT ----------
void updateOutputs() {
  digitalWrite(STATUS_LED_PIN, (buttonPressed && pwmReady) ? HIGH : LOW);
  if (pwmReady) {
    ledcWrite(PWM_LED_PIN, appliedDuty);
  }
}
